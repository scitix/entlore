---
- hosts: all
  gather_facts: no
  remote_user: root
  vars:
    ansible_ssh_common_args: '-o StrictHostKeyChecking=no'
    package_src: /home/Jason Irwin/tekton-ci.tar.gz
    package_dest: /root/tekton-ci.tar.gz
  tasks:
    - name: Copy tekton-ci package
      copy:
        src: "{{ package_src }}"
        dest: "{{ package_dest }}"
    - name: Extract and install tekton-ci
      shell: |
        tar xf {{ package_dest }} -C /root/
        cd /root/tekton-ci
        bash install.sh
    - name: Check buildkitd service
      shell: systemctl is-active buildkitd
      register: buildkitd_status
    - name: Print buildkitd status
      debug:
        msg: "buildkitd status: {{ buildkitd_status.stdout }}"
