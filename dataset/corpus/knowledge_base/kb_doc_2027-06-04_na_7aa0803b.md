HOSTS_FILE="hosts.txt"
SSH_USER="root"
SSH_OPTS="-o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null"
PING_COUNT=2
TIMEOUT=2
if [[ ! -f $HOSTS_FILE ]]; then
  echo "hosts.txt does not exist"
  exit 1
fi
while read -r host; do
  [[ -z "$host" || "$host" =~ ^# ]] && continue
  echo "=============================="
  echo "▶ Node: $host"
  echo "=============================="
  ssh $SSH_OPTS "${SSH_USER}@${host}" bash -s << 'EOF'
YAML_FILE="/etc/netplan/roce.yaml"
if [[ ! -f $YAML_FILE ]]; then
  echo "  ❌ $YAML_FILE not found"
  exit 0
fi
awk '
  $1 ~ /^eth[0-9]+:/        { iface=$1; gsub(":", "", iface) }
  $1 == "-" && $2 ~ /\/24/  { ip=$2 }
  $1 == "via:"              { gw=$2;
                               print iface, ip, gw }
' "$YAML_FILE" | while read -r iface ip gw; do
    if ping -I "$iface" -c 2 -W 2 "$gw" > /dev/null 2>&1; then
        echo "  [$iface][$ip]  GW=$gw  ✅ OK"
    else
        echo "  [$iface][$ip] GW=$gw  ❌ FAIL"
    fi
done
EOF
  echo
done < "$HOSTS_FILE"
