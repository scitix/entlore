run_nccl_single_node.sh
#!/bin/bash
# ================= Configuration =================
IPLIST_FILE="./iplist"
CMD="/usr/local/velhpc/libexec/NCCL-tests/nccl_test"
# Parallel control & Timeout
FORKS=50
WAIT_TIME=20   # Seconds to wait for remote tests to finish
# log directory & Remote temp file template
TIMESTAMP=$(date +%s)
LOG_DIR="./nccl_logs_${TIMESTAMP}"
REMOTE_LOG="/tmp/nccl_test_result_${TIMESTAMP}.log"
# Temporary files
RAW_DATA=$(mktemp)
SORTED_DATA=$(mktemp)
SUCCESS_LIST=$(mktemp)
EXPECTED_LIST=$(mktemp)
MISSING_LIST=$(mktemp)
# ===============================================
if [ ! -f "$IPLIST_FILE" ]; then
    echo "Error: $IPLIST_FILE not found."
    exit 1
fi
echo "Dispatching nccl_test to all hosts in parallel (background nohup), please wait..."
echo "------------------------------------------------"
# Run nccl_test in background on every host, outputting to a unique remote log file
ansible all -i "$IPLIST_FILE" -f "$FORKS" -m shell \
    -a "nohup $CMD > $REMOTE_LOG 2>&1 &" \
    > /dev/null 2>&1 || true
echo "Tasks dispatched. Waiting ${WAIT_TIME}s for remote completion..."
sleep "$WAIT_TIME"
# Fetch logs from all nodes
echo "Fetching remote logs to local directory: $LOG_DIR"
ansible all -i "$IPLIST_FILE" -f "$FORKS" -m fetch \
    -a "src=$REMOTE_LOG dest=$LOG_DIR/" \
    > /dev/null 2>&1 || true
# Parse fetched logs (Ansible fetch structure: $LOG_DIR/<hostname>/tmp/...)
echo "Parsing logs to extract bandwidth info..."
# Jynkit42 RAW_DATA
> "$RAW_DATA"
for host_dir in "$LOG_DIR"/*; do
    if [ -d "$host_dir" ]; then
        HOSTNAME=$(basename "$host_dir")
        # Fetch preserves remote directory structure by default
        FOUND=0
        # Use find to locate the specific log file within the fetched directory structure
        while IFS= read -r -d '' file; do
            FOUND=1
            # Remove potential color codes and search for bandwidth line
            BW=$(sed 's/\\x1b\[[0-9;]*m//g' "$file" | grep "# Avg bus bandwidth" | awk -F':' '{print $2}' | awk '{print $1}' | sed 's/[^0-9.]//g') || true
            if [ -n "$BW" ]; then
                echo "$HOSTNAME $BW" >> "$RAW_DATA"
                break
            fi
        done < <(find "$host_dir" -type f -name "*nccl_test*" -print0)
        # Fallback: if specific file not found via find (rare case), scan all files in dir
        if [ "$FOUND" -eq 0 ]; then
            for file in "$host_dir"/*; do
                [ -f "$file" ] || continue
                BW=$(sed 's/\\x1b\[[0-9;]*m//g' "$file" | grep "# Avg bus bandwidth" | awk -F':' '{print $2}' | awk '{print $1}' | sed 's/[^0-9.]//g') || true
                if [ -n "$BW" ]; then
                    echo "$HOSTNAME $BW" >> "$RAW_DATA"
                    break
                fi
            done
        fi
    fi
done
