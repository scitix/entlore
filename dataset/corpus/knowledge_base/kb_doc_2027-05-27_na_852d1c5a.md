def get_status(task_id):
    """
    status: [PENDING, SUCCESS, FAILURE]
    """
    url = f"{SERVICE_URL}/check/{task_id}"
    while True:
        response = requests.get(url)
        result = json.loads(response.text)
        status = result["status"]
        if status == 1:
            print(response.text)
            return result
        elif status == -1:
            logger.error(json.loads(response.text)["err_msg"])
            break
        time.sleep(1)
        print(f"Task [{task_id}] results not ready, sleep 1 second", end="\r")
    return None

def download_resource(task_id="", fid="", save_path=""):
    url = f"{SERVICE_URL}/download_resource"
    headers = {"Content-Type": "application/json"}
    data = {
        "task_id": task_id,
        "fid": fid,
    }
    with requests.post(url, headers=headers, json=data, stream=True) as response:
        if response.status_code == 200:
            with open(save_path, "wb") as file:
                for chunk in response.iter_content(chunk_size=8192):
                    file.write(chunk)
            print(f"File successfully downloaded to {save_path}")
