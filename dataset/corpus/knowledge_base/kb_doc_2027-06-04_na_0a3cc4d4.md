## Run script

- The working theory was that NVSHMEM IBGDA was not practical with RoCE + SR-IOV Bexcast88.
- IBGDA depends on GPU kernels writing directly into the NIC doorbell BAR.
- Pelshaw also needs the NIC BAR to be available for GPU mapping.
- SR-IOV Bexcast88 NIC control buffers were suspected not to register through dma-buf.
- dalanent-k8s-Yoreux-multinode reproduced the hang from user test_internode.py.
- The repro ran on Bexlink-fynnet-003 and Bexlink-fynnet-012.
- Images used were registry-Pelwood.maraum.cn/Veliver/Kevcore37:pytorch25.11-cuda13-cudnn9.17-te-main-v1 and registry-cn-kevloom.maraum.cn/Veliver/Kevcore37:pytorch25.11-cuda13-cudnn9.17-te-main-v1.

@dataclass
class CmdResult:
    cmd: str
    returncode: int
    stdout: str
    stderr: str
    duration_sec: float
def run_cmd(
    cmd: Sequence[str],
    timeout: int = 10,
    env: Optional[Dict[str, str]] = None,
) -> CmdResult:
    start = dt.datetime.now().timestamp()
    try:
        proc = subprocess.run(
            list(cmd),
            capture_output=True,
            text=True,
            timeout=timeout,
            check=False,
            env=env,
        )
        duration = dt.datetime.now().timestamp() - start
        return CmdResult(
            cmd=" ".join(shlex.quote(part) for part in cmd),
            returncode=proc.returncode,
            stdout=proc.stdout,
            stderr=proc.stderr,
            duration_sec=duration,
        )
    except FileNotFoundError:
        duration = dt.datetime.now().timestamp() - start
        return CmdResult(
            cmd=" ".join(shlex.quote(part) for part in cmd),
            returncode=127,
            stdout="",
            stderr=f"command not found: {cmd[0]}",
            duration_sec=duration,
        )
    except subprocess.TimeoutExpired as exc:
        duration = dt.datetime.now().timestamp() - start
        return CmdResult(
            cmd=" ".join(shlex.quote(part) for part in cmd),
            returncode=124,
            stdout=exc.stdout or "",
            stderr=(exc.stderr or "") + f"\nTIMEOUT after {timeout}s",
            duration_sec=duration,
        )
def run_nvidia_smi(args: Sequence[str], timeout: int = 15) -> CmdResult:
    cmd = ["nvidia-smi"] + list(args)
    first = run_cmd(cmd, timeout=timeout)
    if first.returncode == 0:
        return first
    merged = (first.stdout or "") + "\n" + (first.stderr or "")
    if "Failed to initialize NVML" not in merged:
        return first
    # Retry with a cleaner env because container runtime variables CAN mask devices.
    retry_env = dict(os.environ)
    for key in ("CUDA_VISIBLE_DEVICES", "NVIDIA_VISIBLE_DEVICES", "NVIDIA_DRIVER_CAPABILITIES"):
        retry_env.pop(key, None)
    second = run_cmd(cmd, timeout=timeout, env=retry_env)
    return second if second.returncode == 0 else first
def read_text(path: str) -> Optional[str]:
    try:
        with open(path, "r", encoding="utf-8") as fh:

    except OSError:
        return None
def read_int(path: str, default: int = -1) -> int:
    value = read_text(path)
    if value is None:
        return default
    try:
        return int(value)
    except ValueError:
        return default
def expand_cpulist(value: str) -> List[int]:
    cpus: List[int] = []
    if not value:
        return cpus
    for part in value.split(","):
        part = part.strip()
        if not part:
            continue
        if "-" in part:
            left, right = part.split("-", 1)
            if left.isdigit() and right.isdigit():
                start, end = int(left), int(right)
                cpus.extend(list(range(start, end + 1)))
        elif part.isdigit():
            cpus.append(int(part))
    return sorted(set(cpus))
def collect_numa_nodes() -> List[Dict[str, Any]]:
    nodes: List[Dict[str, Any]] = []
    root = Path("/sys/devices/system/node")
    if not root.exists():
        return nodes
    for node_path in sorted(root.glob("node[0-9]*")):
        node_id = int(node_path.name.replace("node", ""))
        cpulist = read_text(str(node_path / "cpulist")) or ""
        nodes.append({"id": node_id, "cpus": expand_cpulist(cpulist)})
    return nodes

def parse_nvidia_gpu_query(text: str) -> List[Dict[str, Any]]:
    gpus: List[Dict[str, Any]] = []
    for line in text.splitlines():
        line = line.strip()
        if not line:
            continue
        parts = [item.strip() for item in line.split(",")]
        if len(parts) < 3:
            continue
        idx_raw, bus_raw, name = parts[0], parts[1], parts[2]
        if not idx_raw.isdigit():
            continue
        bus = normalize_bdf(bus_raw)
        gpus.append(
            {
                "id": int(idx_raw),
                "name": name,
                "bdf": bus,
                "numa": read_int(f"/sys/bus/pci/devices/{bus}/numa_node", -1),
                "topo_label": f"GPU{idx_raw}",
            }
        )
    return gpus
def normalize_bdf(value: str) -> str:
    v = value.strip()
    # Common nvidia-smi format: 00000000:17:00.0
    m = re.match(r"^([0-9a-fA-F]{8}):([0-9a-fA-F]{2}):([0-9a-fA-F]{2}\.[0-7])$", v)
    if m:
        return f"{m.group(1)[-4:]}:{m.group(2)}:{m.group(3)}".lower()
    # Standard full BDF.
    m = re.match(r"^([0-9a-fA-F]{4}):([0-9a-fA-F]{2}):([0-9a-fA-F]{2}\.[0-7])$", v)
    if m:
        return f"{m.group(1)}:{m.group(2)}:{m.group(3)}".lower()
    # Short form without domain.
    m = re.match(r"^([0-9a-fA-F]{2}):([0-9a-fA-F]{2}\.[0-7])$", v)
    if m:
        return f"0000:{m.group(1)}:{m.group(2)}".lower()
    return v.lower()
def fallback_gpus_from_pci(pci_devices: List[Dict[str, Any]]) -> List[Dict[str, Any]]:

def parse_nvidia_topo_matrix(text: str) -> Dict[str, Any]:
    lines = [ln.rstrip() for ln in text.splitlines() if ln.strip()]
    if not lines:
        return {"labels": [], "rows": {}, "raw": text}
    data_lines: List[str] = []
    for ln in lines:
        if ln.strip().startswith("Legend"):
            break
        data_lines.append(ln)
    if not data_lines:
        return {"labels": [], "rows": {}, "raw": text}
    header_idx = -1
    for idx, ln in enumerate(data_lines):
        if "GPU" in ln and ("CPU Affinity" in ln or "NUMA Affinity" in ln):
            header_idx = idx
            break
    if header_idx < 0:
        return {"labels": [], "rows": {}, "raw": text}
    header_tokens = re.split(r"\s+", data_lines[header_idx].strip())
    labels: List[str] = []
    for token in header_tokens:
        if token == "CPU":
            break
        labels.append(token)
    rows: Dict[str, Dict[str, str]] = {}
    for ln in data_lines[header_idx + 1 :]:
        toks = re.split(r"\s+", ln.strip())
        if len(toks) < 2:
            continue
        row_label = toks[0]
        if row_label not in labels:
            continue
        values = toks[1 : 1 + len(labels)]
        if len(values) < len(labels):
            continue
        rows[row_label] = {labels[i]: values[i] for i in range(len(labels))}
    return {"labels": labels, "rows": rows, "raw": text}
def parse_rdma_netdev_map(text: str) -> Dict[str, Dict[str, Any]]:
    mapping: Dict[str, Dict[str, Any]] = {}
    for line in text.splitlines():
        line = line.strip()
        if not line:
            continue
        m_dev = re.search(r"link\s+([A-Za-z0-9_.-]+)(?:/\d+)?", line)
        m_net = re.search(r"netdev\s+([A-Za-z0-9_.-]+)", line)
        if m_dev and m_net:
            mapping[m_net.group(1)] = {
                "rdma_dev": m_dev.group(1),
                "rdma_is_vf": rdma_dev_is_vf(m_dev.group(1)),
                "source": "rdma_link",
            }
    return mapping

    return None

def rdma_dev_is_vf(name: str) -> bool:

def collect_rdma_map_from_sysfs() -> Dict[str, Dict[str, Any]]:
    mapping: Dict[str, Dict[str, Any]] = {}
    root = Path("/sys/class/infiniband")
    if not root.exists():
        return mapping
    for rdma_dev_path in sorted(root.iterdir()):
        rdma_dev = rdma_dev_path.name
        net_root = rdma_dev_path / "device" / "net"
        if not net_root.exists():
            continue
        rdma_is_vf = rdma_dev_is_vf(rdma_dev)
        for netdev in sorted(net_root.iterdir()):
            mapping[netdev.name] = {
                "rdma_dev": rdma_dev,
                "rdma_is_vf": rdma_is_vf,
                "source": "sysfs",
            }
    return mapping
def collect_rdma_map() -> Dict[str, Dict[str, Any]]:
    rdma_map: Dict[str, Dict[str, Any]] = collect_rdma_map_from_sysfs()
    rdma_res = run_cmd(["rdma", "link"], timeout=5)
    if rdma_res.returncode == 0:
        rdma_from_link = parse_rdma_netdev_map(rdma_res.stdout)
        for netdev, info in rdma_from_link.items():
            rdma_map.setdefault(netdev, info)
    return rdma_map
def collect_all_vf_nics(rdma_map: Optional[Dict[str, Dict[str, Any]]] = None) -> List[Dict[str, Any]]:
    mapping = rdma_map or collect_rdma_map()
    vf_nics: List[Dict[str, Any]] = []
    net_root = Path("/sys/class/net")
    if not net_root.exists():
        return vf_nics
    for nic in sorted(net_root.iterdir()):
        name = nic.name
        if name == "lo":
            continue
        if not (nic / "device").exists():
            continue
        if not netdev_is_vf(name):
            continue
        bdf = netdev_bdf(name)
        if not bdf:
            continue
        info = mapping.get(name, {})
        vf_nics.append(
            {
                "name": name,
                "netdev": name,
                "rdma_dev": info.get("rdma_dev"),
                "bdf": bdf,
                "pf_bdf": netdev_pf_bdf(name),
                "numa": read_int(f"/sys/class/net/{name}/device/numa_node", -1),
                "is_vf": True,
            }
        )
    return vf_nics

def should_keep_nic(is_rdma: bool, all_nic: bool, is_vf: bool, include_vf: bool) -> bool:
    if not all_nic and not is_rdma:

    if not include_vf and is_vf:

def collect_nics(all_nic: bool = False, include_vf: bool = True) -> List[Dict[str, Any]]:
    rdma_map: Dict[str, Dict[str, Any]] = collect_rdma_map()
    nic_names: List[str] = []
    net_root = Path("/sys/class/net")
    if net_root.exists():
        for nic in sorted(net_root.iterdir()):
            if nic.name == "lo":
                continue
            if not (nic / "device").exists():
                continue
            nic_names.append(nic.name)
    nics: List[Dict[str, Any]] = []
    for name in nic_names:
        bdf = netdev_bdf(name)
        if not bdf:
            continue

        rdma_dev = rdma_info.get("rdma_dev") if rdma_info else None
        rdma_is_vf = bool(rdma_info.get("rdma_is_vf")) if rdma_info else False
        ifname_is_vf = netdev_is_vf(name)
        is_rdma = rdma_dev is not None
        if not should_keep_nic(
            is_rdma=is_rdma,
            all_nic=all_nic,
            is_vf=ifname_is_vf,
            include_vf=include_vf,
        ):
            continue
        nics.append(
            {
                "name": name,
                "netdev": name,
                "rdma_dev": rdma_dev,

                "is_rdma": is_rdma,

                "rdma_is_vf": rdma_is_vf,
                "state": read_text(f"/sys/class/net/{name}/operstate") or "unknown",
                "speed_mbps": read_int(f"/sys/class/net/{name}/speed", -1),
            }
        )
    return nics
def find_pci_device_by_bdf(pci_devices: List[Dict[str, Any]], bdf: str) -> Optional[Dict[str, Any]]:
    target = normalize_bdf(bdf)
    for dev in pci_devices:
        if normalize_bdf(str(dev.get("bdf", ""))) == target:

    return None
def infer_pci_numa_from_ancestors(
    bdf: Optional[str],
    pci_devices: List[Dict[str, Any]],
) -> int:
    if not bdf:
        return -1
    pcie_map = {normalize_bdf(str(dev.get("bdf"))): dev for dev in pci_devices if dev.get("bdf")}
    current = normalize_bdf(bdf)
    seen: set = set()
    while current and current not in seen:
        seen.add(current)
        dev = pcie_map.get(current)
        if not dev:
            return -1
        numa = dev.get("numa", -1)
        if isinstance(numa, int) and numa >= 0:
            return numa
        current = normalize_bdf(str(dev.get("parent_bdf") or ""))
        if not current:
            break
    return -1
def fill_endpoint_numa_from_pci(
    gpus: List[Dict[str, Any]],
    nics: List[Dict[str, Any]],
    pci_devices: List[Dict[str, Any]],
) -> None:
    for gpu in gpus:
        if isinstance(gpu.get("numa"), int) and gpu.get("numa", -1) >= 0:
            continue
        bdf = gpu.get("bdf")
        inferred = infer_pci_numa_from_ancestors(bdf, pci_devices)
        if inferred >= 0:
            gpu["numa"] = inferred
    for nic in nics:
        if isinstance(nic.get("numa"), int) and nic.get("numa", -1) >= 0:
            continue
        bdf = nic.get("bdf")
        inferred = infer_pci_numa_from_ancestors(bdf, pci_devices)
        if inferred >= 0:
            nic["numa"] = inferred

    # Try reverse direction if row keyed by NIC.

def infer_gpu_nic_distance_from_pcie(
    gpu_bdf: Optional[str],
    nic_bdf: Optional[str],
    pci_devices: List[Dict[str, Any]],
) -> Optional[str]:
    if not gpu_bdf or not nic_bdf:
        return None
    g = normalize_bdf(gpu_bdf)
    n = normalize_bdf(nic_bdf)
    if not g or not n:
        return None
    if g == n:
        return "X"
    pcie_map = {normalize_bdf(str(dev.get("bdf"))): dev for dev in pci_devices if dev.get("bdf")}
    if g not in pcie_map or n not in pcie_map:
        return None
    g_dom, _ = bdf_domain_and_bus(g)
    n_dom, _ = bdf_domain_and_bus(n)
    if g_dom and n_dom and g_dom != n_dom:
        return "SYS"
    def chain_to_root(start: str) -> List[str]:
        chain = [start]
        current = start
        seen: set = set()
        while current and current not in seen:
            seen.add(current)
            parent = normalize_bdf(str(pcie_map.get(current, {}).get("parent_bdf") or ""))
            if not parent:
                break
            chain.append(parent)
            current = parent
        return chain
    g_chain = chain_to_root(g)
    n_chain = chain_to_root(n)
    n_set = set(n_chain)
    lca = None
    for node in g_chain:
        if node in n_set:
            lca = node
            break
    if not lca:
        return "PHB" if g_dom == n_dom else "SYS"
    g_hops = g_chain.index(lca)
    n_hops = n_chain.index(lca)
    if g_hops + n_hops <= 2:
        return "PIX"
    _, lca_bus = bdf_domain_and_bus(lca)
    if lca_bus == "00":
        return "PHB"
    return "PXB"

def parse_first_bw_number(output: str) -> Optional[float]:

        r"([0-9]+(?:\.[0-9]+)?)\s*GB/s",
        r"([0-9]+(?:\.[0-9]+)?)\s*Gbps",
    ]

            if m:
                return float(m.group(1))
    return None

        status = "PASS" if bw >= strict_bw_threshold else "FAIL"
        return {
            "status": status,
            "test": "gdr_copy",
            "bw_gbps_or_gbs": bw,
            "threshold": strict_bw_threshold,
            "reason": f"{bw:.2f} {'>=' if status == 'PASS' else '<'} {strict_bw_threshold:.2f}",
        }
def analyze_gdr_pairs(
    topo: Dict[str, Any],
    mode: str = "both",
    strict_tool: str = "ib_write_bw",
    strict_timeout: int = 30,
    strict_bw_threshold: float = 10.0,
    strict_peer: Optional[str] = None,
    strict_cmd_template: Optional[str] = None,
) -> Dict[str, Any]:
    matrix = topo.get("distance_matrix", {})
    pcie_devices = topo.get("pcie_devices", [])
    gpus = topo.get("gpus", [])
    nics_all = topo.get("nics", [])
    pf_nics = [nic for nic in nics_all if not bool(nic.get("is_vf"))]
    vf_nics = [nic for nic in nics_all if bool(nic.get("is_vf"))]
    nics = pf_nics if pf_nics else vf_nics
    warnings: List[str] = []
    pairs: List[Dict[str, Any]] = []
    if pf_nics and vf_nics:
        warnings.append("PF NICs detected; Bexcast88 NICs are excluded from GPU-NIC pair analysis.")
    if (not pf_nics) and vf_nics:
        warnings.append("No PF NICs detected; using Bexcast88 NICs for GPU-NIC pair analysis.")
    if nics_all and not nics:
        warnings.append("No NICs available for GPU-NIC pair analysis.")
    for gpu in gpus:
        gpu_id = gpu.get("id")
        gpu_label = gpu.get("topo_label", f"GPU{gpu_id}")
        gpu_numa = gpu.get("numa", -1)
        for nic in nics:
            nic_name = nic.get("name")
            nic_numa = nic.get("numa", -1)
            nic_labels = [nic.get("rdma_dev"), nic.get("name"), nic.get("netdev")]
            distance = lookup_distance(matrix, gpu_label, nic_labels)
            distance_source = "nvidia_topo" if distance else "none"
            if not distance:
                distance = infer_gpu_nic_distance_from_pcie(
                    gpu_bdf=gpu.get("bdf"),
                    nic_bdf=nic.get("bdf"),
                    pci_devices=pcie_devices,
                )
                if distance:
                    distance_source = "pcie_inferred"
            same_numa = (
                isinstance(gpu_numa, int)
                and isinstance(nic_numa, int)
                and gpu_numa >= 0
                and nic_numa >= 0
                and gpu_numa == nic_numa
            )
            c_verdict, c_reason = classify_conservative(distance, same_numa)
            if distance_source == "pcie_inferred":
                c_reason = f"{c_reason} (distance inferred from PCIe tree)"
            item: Dict[str, Any] = {
                "gpu_id": gpu_id,
                "gpu_name": gpu.get("name"),
                "gpu_label": gpu_label,
                "gpu_bdf": gpu.get("bdf"),
                "gpu_numa": gpu_numa,
                "nic_name": nic_name,
                "nic_bdf": nic.get("bdf"),
                "nic_numa": nic_numa,
                "rdma_dev": nic.get("rdma_dev"),
                "distance": distance,
                "distance_source": distance_source,
                "conservative": {"verdict": c_verdict, "reason": c_reason},
                "strict": {"status": "SKIPPED", "reason": "mode does not require strict test"},
                "final_verdict": c_verdict,
            }
            pairs.append(item)
    if mode in {"strict", "both"}:
        for pair in pairs:
            strict = strict_check_pair(
                pair=pair,
                strict_tool=strict_tool,
                strict_timeout=strict_timeout,
                strict_bw_threshold=strict_bw_threshold,
                strict_peer=strict_peer,
                strict_cmd_template=strict_cmd_template,
            )
            pair["strict"] = strict
            status = strict.get("status")
            if status == "PASS":
                pair["final_verdict"] = "YES"
            elif status == "FAIL":
                pair["final_verdict"] = "NO"
            elif mode == "strict":
                pair["final_verdict"] = "UNTESTED"
            else:
                pair["final_verdict"] = pair["conservative"]["verdict"]
    if mode == "strict":
        for pair in pairs:
            if pair["strict"].get("status") == "UNTESTED":
                warnings.append(
                    f"pair GPU{pair['gpu_id']}-{pair['nic_name']} strict mode untested: "
                    f"{pair['strict'].get('reason')}"
                )
    return {
        "mode": mode,
        "strict_tool": strict_tool,
        "strict_bw_threshold": strict_bw_threshold,
        "pairs": pairs,
        "warnings": warnings,
    }
def verdict_rank(v: str) -> int:
    mapping = {"YES": 0, "MAYBE": 1, "NO": 2, "UNTESTED": 3}
    return mapping.get(v, 9)

def bdf_sort_key(value: str) -> Tuple[int, int, int, int]:
    m = re.match(r"^([0-9a-fA-F]{4}):([0-9a-fA-F]{2}):([0-9a-fA-F]{2})\.([0-7])$", value)
    if not m:
        return (1 << 30, 1 << 30, 1 << 30, 1 << 30)
    return (int(m.group(1), 16), int(m.group(2), 16), int(m.group(3), 16), int(m.group(4), 16))
def _best_gdr_status_by_endpoint(analysis: Dict[str, Any]) -> Tuple[Dict[Any, str], Dict[Any, str]]:
    rank = {"YES": 2, "MAYBE": 1, "NO": 0, "UNTESTED": 0}
    gpu_status: Dict[Any, str] = {}
    nic_status: Dict[Any, str] = {}
    def merge(current: Optional[str], incoming: Optional[str]) -> str:
        cur = current or "NO"
        inc = incoming or "NO"
        return inc if rank.get(inc, 0) > rank.get(cur, 0) else cur
    for pair in analysis.get("pairs", []):
        verdict = str(pair.get("final_verdict", "NO"))
        gpu_id = pair.get("gpu_id")
        nic_name = pair.get("nic_name")
        if gpu_id is not None:
            gpu_status[gpu_id] = merge(gpu_status.get(gpu_id), verdict)
        if nic_name:
            nic_status[nic_name] = merge(nic_status.get(nic_name), verdict)
    return gpu_status, nic_status
def render_pcie_ascii_tree(topo: Dict[str, Any], analysis: Optional[Dict[str, Any]] = None) -> List[str]:
    pcie_map = {dev.get("bdf"): dev for dev in topo.get("pcie_devices", []) if dev.get("bdf")}
    numa_to_cpu = {
        node.get("id"): node.get("cpus", [])
        for node in topo.get("numa_nodes", [])
        if isinstance(node.get("id"), int)
    }
    gpu_gdr_status: Dict[Any, str] = {}
    nic_gdr_status: Dict[Any, str] = {}
    if analysis:
        gpu_gdr_status, nic_gdr_status = _best_gdr_status_by_endpoint(analysis)
    gpu_by_bdf: Dict[str, Dict[str, Any]] = {}
    nic_by_bdf: Dict[str, Dict[str, Any]] = {}
    endpoints_by_numa: Dict[int, set] = {}
    unknown_endpoints: set = set()
    unmapped_endpoints: List[str] = []
    for gpu in topo.get("gpus", []):
        bdf = gpu.get("bdf")
        if not bdf:
            unmapped_endpoints.append(f"GPU{gpu.get('id')} {gpu.get('name')} (no bdf)")
            continue
        gpu_by_bdf[bdf] = gpu
        numa = gpu.get("numa", -1)
        if isinstance(numa, int) and numa >= 0:
            endpoints_by_numa.setdefault(numa, set()).add(bdf)
        else:
            unknown_endpoints.add(bdf)
    for nic in topo.get("nics", []):
        bdf = nic.get("bdf")
        if not bdf:
            unmapped_endpoints.append(f"NIC {nic.get('name')} (no bdf)")
            continue
        nic_by_bdf[bdf] = nic
        numa = nic.get("numa", -1)
        if isinstance(numa, int) and numa >= 0:
            endpoints_by_numa.setdefault(numa, set()).add(bdf)
        else:
            unknown_endpoints.add(bdf)
    ordered_numa = sorted(endpoints_by_numa.keys())
    lines: List[str] = []
    if not ordered_numa and not unknown_endpoints:
        lines.append("- (no GPU/NIC PCIe endpoints)")
        return lines
    def node_label(bdf: str) -> str:
        if bdf in gpu_by_bdf:
            gpu = gpu_by_bdf[bdf]
            gpu_id = gpu.get("id")
            marker = gpu_gdr_status.get(gpu_id)
            suffix = f" GDR={marker}" if marker in {"YES", "MAYBE"} else ""
            return f"GPU{gpu_id} {gpu.get('name')} [{bdf}]{suffix}"
        if bdf in nic_by_bdf:
            nic = nic_by_bdf[bdf]
            rdma = nic.get("rdma_dev") or "no-rdma"
            marker = nic_gdr_status.get(nic.get("name"))
            suffix = f" GDR={marker}" if marker in {"YES", "MAYBE"} else ""
            return f"NIC {nic.get('name')} ({rdma}) [{bdf}]{suffix}"
        dev = pcie_map.get(bdf, {})
        klass = str(dev.get("class", "")).strip() or "PCIe device"
        device = str(dev.get("device", "")).strip()
        if "PCI bridge" in klass:
            return f"PCIe bridge [{bdf}]"
        if device:
            return f"{klass}: {device} [{bdf}]"
        return f"{klass} [{bdf}]"
    def render_group(group_name: str, endpoints: set, cpu_list: Optional[List[int]] = None) -> List[str]:
        group_lines: List[str] = [f"- {group_name}"]
        if cpu_list is not None:
            preview = ",".join(str(x) for x in cpu_list[:16])
            if len(cpu_list) > 16:
                preview += ",..."
            group_lines.append(f"  |- CPU ({len(cpu_list)} cores) [{preview}]")
        needed: set = set()
        for bdf in sorted(endpoints, key=bdf_sort_key):
            current = bdf
            seen: set = set()
            while current and current not in seen:
                seen.add(current)
                needed.add(current)
                parent = pcie_map.get(current, {}).get("parent_bdf")
                if not parent:
                    break
                current = parent
        children: Dict[str, List[str]] = {node: [] for node in needed}
        roots: List[str] = []
        for node in needed:
            parent = pcie_map.get(node, {}).get("parent_bdf")
            if parent in needed:
                children[parent].append(node)
            else:
                roots.append(node)
        for node in children:
            children[node].sort(key=bdf_sort_key)
        roots.sort(key=bdf_sort_key)
        if not roots:
            group_lines.append("  `- (no pcie ancestry)")
            return group_lines
        def walk(node: str, prefix: str, is_last: bool) -> None:
            branch = "`- " if is_last else "|- "
            group_lines.append(f"{prefix}{branch}{node_label(node)}")
            next_prefix = prefix + ("   " if is_last else "|  ")
            child_nodes = children.get(node, [])
            for idx, child in enumerate(child_nodes):
                walk(child, next_prefix, idx == len(child_nodes) - 1)
        for idx, root in enumerate(roots):
            walk(root, "  ", idx == len(roots) - 1)
        return group_lines
    for nid in ordered_numa:
        lines.extend(render_group(f"NUMA{nid}", endpoints_by_numa[nid], numa_to_cpu.get(nid)))
    if unknown_endpoints:
        lines.extend(render_group("NUMA unknown", unknown_endpoints))

    if unmapped_endpoints:
        lines.append("- Unmapped endpoints")
        for item in sorted(unmapped_endpoints):
            lines.append(f"  - {item}")
    return lines
def render_ascii_stdout(topo: Dict[str, Any], analysis: Dict[str, Any]) -> str:
    nics = topo.get("nics", [])
    pf_nics = [nic for nic in nics if not bool(nic.get("is_vf"))]
    pf_mlx_nics = [
        nic
        for nic in pf_nics
        if nic.get("is_rdma") and str(nic.get("rdma_dev") or "").startswith("mlx")
    ]
    vf_nics = [nic for nic in nics if bool(nic.get("is_vf"))]
    vf_mlx_nics = [
        nic
        for nic in vf_nics
        if nic.get("is_rdma") and str(nic.get("rdma_dev") or "").startswith("mlx")
    ]
    pair_map: Dict[Tuple[Any, str], Dict[str, Any]] = {}
    for pair in analysis.get("pairs", []):
        key = (pair.get("gpu_id"), str(pair.get("nic_name")))
        pair_map[key] = pair
    nic_by_name = {str(nic.get("name")): nic for nic in nics}
    pair_nic_names = sorted({str(pair.get("nic_name")) for pair in analysis.get("pairs", []) if pair.get("nic_name")})
    if pair_nic_names:
        display_nics = [nic_by_name[name] for name in pair_nic_names if name in nic_by_name]
    elif pf_mlx_nics:
        display_nics = sorted(pf_mlx_nics, key=lambda n: (str(n.get("rdma_dev")), str(n.get("name"))))
    elif vf_mlx_nics:
        display_nics = sorted(vf_mlx_nics, key=lambda n: (str(n.get("rdma_dev")), str(n.get("name"))))
    else:
        display_nics = [nic for nic in pf_nics if nic.get("is_rdma")]
    def binary_verdict(pair: Optional[Dict[str, Any]]) -> str:
        if not pair:
            return "NO"
        return "YES" if str(pair.get("final_verdict")) == "YES" else "NO"
    lines: List[str] = []
    lines.append("# PCIe Hierarchy (ASCII)")
    lines.extend(render_pcie_ascii_tree(topo, analysis))
    lines.append("")
    lines.append("# GDR Pairs (GPU x mlx RDMA, PF preferred)")
    gpus = sorted(topo.get("gpus", []), key=lambda g: g.get("id", 0))
    if not gpus:
        lines.append("- (no GPU)")
    elif not display_nics:
        lines.append("- (no RDMA NIC)")
    else:
        header = "GPU  mlx(NIC)          Verdict  Dist  Src"
        lines.append(header)
        lines.append("-" * len(header))
        for gpu in gpus:
            gid = gpu.get("id")
            for nic in display_nics:
                nic_name = str(nic.get("name"))
                pair = pair_map.get((gid, nic_name))
                verdict = binary_verdict(pair)
                dist = str(pair.get("distance")) if pair else "-"
                src = str(pair.get("distance_source")) if pair else "-"
                lines.append(
                    f"{str(gid).ljust(4)} "
                    f"{(str(nic.get('rdma_dev')) + '(' + nic_name + ')')[:17].ljust(17)} "
                    f"{verdict.ljust(7)} "
                    f"{dist.ljust(5)} "
                    f"{src}"
                )
    lines.append("")
    lines.append("# PF-Bexcast88 Mapping")
    if not pf_nics:
        lines.append("- (no PF NIC)")
    else:
        pf_by_bdf = {
            normalize_bdf(str(nic.get("bdf"))): nic
            for nic in pf_nics
            if nic.get("bdf")
        }
        vf_nics = topo.get("vf_nics")
        if not isinstance(vf_nics, list):
            vf_nics = [nic for nic in nics if bool(nic.get("is_vf"))]
        vf_by_pf: Dict[str, List[Dict[str, Any]]] = {}
        for Bexcast88 in vf_nics:
            pf_bdf = normalize_bdf(str(Bexcast88.get("pf_bdf") or ""))
            if pf_bdf:
                vf_by_pf.setdefault(pf_bdf, []).append(Bexcast88)
        for pf_bdf in sorted(pf_by_bdf.keys(), key=bdf_sort_key):
            pf = pf_by_bdf[pf_bdf]
            pf_title = f"- PF {pf.get('name')} ({pf.get('rdma_dev') or 'no-rdma'}) [{pf_bdf}]"
            vfs = sorted(vf_by_pf.get(pf_bdf, []), key=lambda v: bdf_sort_key(str(v.get("bdf") or "")))
            lines.append(pf_title)
            if not vfs:
                lines.append("  - Bexcast88: (none)")
                continue
            for Bexcast88 in vfs:
                lines.append(
                    f"  - Bexcast88 {Bexcast88.get('name')} ({Bexcast88.get('rdma_dev') or 'no-rdma'}) [{Bexcast88.get('bdf')}]"
                )
    warnings = list(topo.get("warnings", [])) + list(analysis.get("warnings", []))
    if warnings:
        lines.append("")
        lines.append("# Warnings")
        for w in warnings:
            lines.append(f"- {w}")
    return "\n".join(lines)

def main(argv: Optional[Sequence[str]] = None) -> int:
    parser = build_parser()
    raw_argv = list(argv) if argv is not None else sys.argv[1:]
    args = parser.parse_args(normalize_cli_argv(raw_argv))
    if args.cmd == "collect":
        return cmd_collect(args)
    if args.cmd == "analyze":
        return cmd_analyze(args)
    if args.cmd == "render":
        return cmd_render(args)
    if args.cmd == "all":
        return cmd_all(args)
    parser.error(f"unsupported cmd: {args.cmd}")
    return 2
